    <script>
        // Mobile Navigation Toggle
        const mobileBtn = document.getElementById('mobile-menu-btn');

        // Countdown Timer Logic (Target Date: April 24, 2027 at 18:30 CDT)
        const targetDate = new Date('April 24, 2027 18:30:00').getTime();

        function updateCountdown() {
            const now = new Date().getTime();
            const difference = targetDate - now;

            if (difference > 0) {
                const days = Math.floor(difference / (1000 * 60 * 60 * 24));
                const hours = Math.floor((difference % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                const minutes = Math.floor((difference % (1000 * 60 * 60)) / (1000 * 60));
                const seconds = Math.floor((difference % (1000 * 60)) / 1000);

                document.getElementById('cd-days').innerText = String(days).padStart(2, '0');
                document.getElementById('cd-hours').innerText = String(hours).padStart(2, '0');
                document.getElementById('cd-minutes').innerText = String(minutes).padStart(2, '0');
                document.getElementById('cd-seconds').innerText = String(seconds).padStart(2, '0');
            } else {
                document.getElementById('cd-days').innerText = '00';
                document.getElementById('cd-hours').innerText = '00';
                document.getElementById('cd-minutes').innerText = '00';
                document.getElementById('cd-seconds').innerText = '00';
            }
        }

        setInterval(updateCountdown, 1000);
        updateCountdown();

        // FAQ Accordion Toggle
        document.querySelectorAll('.faq-toggle').forEach(button => {
            button.addEventListener('click', () => {
                const content = button.nextElementSibling;
                const icon = button.querySelector('i');

                content.classList.toggle('hidden');
                
                if (content.classList.contains('hidden')) {
                    icon.style.transform = 'rotate(0deg)';
                } else {
                    icon.style.transform = 'rotate(180deg)';
                }
            });
        });

        // Copy Address Helper
        function copyAddress() {
            const address = "41048 Aline Blacktop Rd, Aline, OK 73716";
            const copyBtn = document.getElementById('copy-btn');
            
            const tempInput = document.createElement('input');
            tempInput.value = address;
            document.body.appendChild(tempInput);
            tempInput.select();
            document.execCommand('copy');
            document.body.removeChild(tempInput);

            copyBtn.innerHTML = '<i class="fa-solid fa-check text-sage-green mr-1"></i> Address Copied!';
            setTimeout(() => {
                copyBtn.innerHTML = '<i class="fa-regular fa-copy mr-1"></i> Copy Address';
            }, 3000);
        }

        // --- RSVP Local Storage & Dashboard System ---
        const RSVP_STORAGE_KEY = 'brock_karly_wedding_rsvps';

        function getStoredRSVPs() {
            return JSON.parse(localStorage.getItem(RSVP_STORAGE_KEY) || '[]');
        }

        function saveRSVPToLocalStorage(rsvpData) {
            const rsvps = getStoredRSVPs();
            rsvps.unshift(rsvpData);
            localStorage.setItem(RSVP_STORAGE_KEY, JSON.stringify(rsvps));
            renderRSVPDashboard();
        }

        // RSVP Form Submission Handler
        const rsvpForm = document.getElementById('rsvp-form');
        const rsvpSuccess = document.getElementById('rsvp-success');

            const submitBtn = document.getElementById('submit-btn');
            submitBtn.disabled = true;
            submitBtn.innerHTML = '<i class="fa-solid fa-spinner fa-spin mr-2"></i> Submitting...';

            const formData = {
                id: Date.now(),
                timestamp: new Date().toLocaleString('en-US', { dateStyle: 'short', timeStyle: 'short' }),

                attending: document.getElementById('attending').value,

                songRequest: document.getElementById('song-request').value.trim() || 'None',
                note: document.getElementById('guest-note').value.trim() || 'None'
            };

            // Save locally
            saveRSVPToLocalStorage(formData);

            submitBtn.innerHTML = '<span>Submit RSVP</span><i class="fa-solid fa-paper-plane text-sm"></i>';

        function renderRSVPDashboard() {
            const rsvps = getStoredRSVPs();
            const tableBody = document.getElementById('rsvp-table-body');
            
            // Calculate Stats
            let attendingCount = 0;
            let adultCount = 0;
            let childCount = 0;

            rsvps.forEach(r => {
                if (r.attending === 'Yes') {
                    attendingCount++;
                    adultCount += parseInt(r.adults || 0);
                    childCount += parseInt(r.children || 0);
                }
            });

            document.getElementById('stat-total').innerText = rsvps.length;
            document.getElementById('stat-attending').innerText = attendingCount;
            document.getElementById('stat-adults').innerText = adultCount;
            document.getElementById('stat-kids').innerText = childCount;

            if (rsvps.length === 0) {
                tableBody.innerHTML = `

                return;
            }

            tableBody.innerHTML = rsvps.map(r => `
                <tr class="hover:bg-baby-light/40 transition">
                    <td class="p-3 text-slate-dark/60 whitespace-nowrap">${r.timestamp}</td>
                    <td class="p-3 font-semibold text-dusty-dark">${escapeHtml(r.fullName)}</td>
                    <td class="p-3 whitespace-nowrap">
                        <span class="px-2.5 py-1 rounded-full text-[10px] font-bold uppercase ${r.attending === 'Yes' ? 'bg-sage-light text-sage-dark' : 'bg-red-50 text-red-600'}">
                            ${r.attending}
                        </span>
                    </td>
                    <td class="p-3 font-medium">${r.adults}</td>
                    <td class="p-3 font-medium">${r.children}</td>
                    <td class="p-3 text-slate-dark/80">${escapeHtml(r.songRequest)}</td>
                    <td class="p-3 text-slate-dark/80 max-w-xs truncate">${escapeHtml(r.note)}</td>
                </tr>
            `).join('');
        }

        function clearAllRSVPs() {
            if (confirm("Are you sure you want to clear all stored RSVPs? This action cannot be undone.")) {
                localStorage.removeItem(RSVP_STORAGE_KEY);
                renderRSVPDashboard();
            }
        }

        function exportToCSV() {
            const rsvps = getStoredRSVPs();
            if (rsvps.length === 0) {
                alert("No RSVP data available to export yet!");
                return;
            }

            const headers = ["Timestamp", "Guest Name", "Attending", "Adults", "Children", "Song Request", "Message Note"];
            const rows = rsvps.map(r => [
                `"${r.timestamp}"`,
                `"${r.fullName.replace(/"/g, '""')}"`,
                `"${r.attending}"`,
                r.adults,
                r.children,
                `"${r.songRequest.replace(/"/g, '""')}"`,
                `"${r.note.replace(/"/g, '""')}"`
            ]);

            const csvContent = "data:text/csv;charset=utf-8," + [headers.join(","), ...rows.map(e => e.join(","))].join("\n");
            const encodedUri = encodeURI(csvContent);
            
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `Brock_and_Karly_Wedding_RSVPs_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        function escapeHtml(str) {
            if (!str) return '';
            return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;");
        }
